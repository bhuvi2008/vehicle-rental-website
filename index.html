3NF Relational Database Schema & Live State Inspector

TABLE 1 VEHICLES (3NF Entity)

  • PK vehicle_id VARCHAR(12) — e.g., VEH-101
  • model_name VARCHAR(80) NOT NULL
  • category ENUM('Bikes & Scooters','Basic Cars','Sedan','SUV','Electric')
  • daily_rate DECIMAL(10,2) CHECK (daily_rate > 0)
  • license_class VARCHAR(30) — MCWG / LMV
  • status ENUM('AVAILABLE','BOOKED','MAINTENANCE')

TABLE 2 BOOKINGS & PAYMENTS (3NF)

  • PK booking_id VARCHAR(16) — e.g., BKG-4821
  • FK vehicle_id REFERENCES VEHICLES(vehicle_id)
  • customer_name VARCHAR(80), dl_number VARCHAR(25)
  • pickup_date DATE, return_date DATE
  • gst_amount DECIMAL(10,2), grand_total DECIMAL(10,2)
  • booking_status ENUM('ACTIVE','COMPLETED','CANCELLED')
